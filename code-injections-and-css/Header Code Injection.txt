<script>
(function () {
  'use strict';

  var FORM_URL = 'https://www.studentluxe.co.uk/reservations';

  // Valid city values on the /reservations form. Anything not in this list is
  // dropped rather than passed through, so the dropdown never gets a value it
  // cannot select.
  var CITIES = ['amsterdam','barcelona','birmingham','boston','brighton','bristol','cambridge','chicago','dublin','durham','edinburgh','florence','glasgow','helsinki','lisbon','liverpool','london','madrid','manchester','milan','new-york','nottingham','paris','philadelphia','porto','rome','valencia','washington'];

  function normCity(v) {
    if (!v) return '';
    var s = String(v).trim().toLowerCase().replace(/\s+/g, '-');
    if (s === 'nyc' || s === 'new-york-city') s = 'new-york';
    return CITIES.indexOf(s) !== -1 ? s : '';
  }

  // JotForm sends a free-text room type ("Studio", "Deluxe Studio", "2 Bed").
  // The /reservations apt values are studio | 1bed | 2bed | 3bed | 4bed | flexible.
  function normAptType(v) {
    if (!v) return '';
    var s = String(v).toLowerCase();
    if (s.indexOf('studio') !== -1) return 'studio';
    var m = s.match(/(\d)\s*-?\s*bed/);
    if (m) return (m[1] >= '1' && m[1] <= '4') ? m[1] + 'bed' : 'flexible';
    if (/\bone\b/.test(s))   return '1bed';
    if (/\btwo\b/.test(s))   return '2bed';
    if (/\bthree\b/.test(s)) return '3bed';
    if (/\bfour\b/.test(s))  return '4bed';
    return '';
  }

  function isJotform(href) {
    return typeof href === 'string' && href.toLowerCase().indexOf('jotform.com') !== -1;
  }

  // Map an old JotForm URL to the equivalent /reservations URL.
  // The param names deliberately do NOT pass through, they collide:
  //   JotForm apartmentOr / apt  = apartment NAME  -> /reservations ref
  //   JotForm type               = ROOM type       -> /reservations apt
  //   JotForm chooseA            = city            -> /reservations city
  //   JotForm search             = intent blurb    -> dropped
  // A passthrough would put "Deluxe Studio" into type (expects A or B) and an
  // apartment name into apt (expects studio/1bed), breaking the prefill.
  function mapUrl(href) {
    var params = null;
    var qi = href.indexOf('?');
    if (qi !== -1) {
      try { params = new URLSearchParams(href.slice(qi + 1)); } catch (e) { params = null; }
    }
    var get = function (k) { return params ? (params.get(k) || '') : ''; };

    var refName = (get('apartmentOr') || get('apt') || '').trim();
    var city    = normCity(get('chooseA') || get('city'));
    var aptType = normAptType(get('type') || get('roomType'));

    // An apartment name means the guest wants that specific place: type A.
    // No name means the generic "send me options" journey: type B.
    var out = refName
      ? ['type=A', 'ref=' + encodeURIComponent(refName)]
      : ['type=B'];
    if (city)    out.push('city=' + encodeURIComponent(city));
    if (aptType) out.push('apt=' + encodeURIComponent(aptType));

    return FORM_URL + '?' + out.join('&');
  }

  // Pass 1: rewrite every JotForm link currently in the DOM.
  // Iterates all anchors rather than using a[href*="jotform"] because CSS
  // attribute selectors are case sensitive and some links use JotForm.com.
  function rewriteAll() {
    var links;
    try { links = document.querySelectorAll('a[href]'); } catch (e) { return; }
    for (var i = 0; i < links.length; i++) {
      var href = links[i].getAttribute('href');
      if (!isJotform(href)) continue;
      links[i].setAttribute('href', mapUrl(href));
      links[i].removeAttribute('target');
      links[i].removeAttribute('rel');
    }
    var forms;
    try { forms = document.querySelectorAll('form[action]'); } catch (e) { return; }
    for (var j = 0; j < forms.length; j++) {
      var action = forms[j].getAttribute('action');
      if (isJotform(action)) forms[j].setAttribute('action', mapUrl(action));
    }
  }

  // Pass 2: capture-phase click interceptor. This is the guarantee, and the
  // part the old code lacked. It resolves the link at click time, so it works
  // even when the anchor was rendered a moment ago by Squarespace's own JS, or
  // when a re-render (mobile menu, breakpoint change) restored the original
  // href from the header's data-current-styles JSON after pass 1 ran.
  document.addEventListener('click', function (e) {
    var el = e.target;
    while (el && el !== document) {
      if (el.tagName === 'A') {
        var href = el.getAttribute('href');
        if (isJotform(href)) {
          var target = mapUrl(href);
          // Let modified clicks (new tab/window) behave natively, just fix the
          // destination first rather than hijacking them.
          if (e.metaKey || e.ctrlKey || e.shiftKey || e.button !== 0) {
            el.setAttribute('href', target);
            el.removeAttribute('target');
            return;
          }
          e.preventDefault();
          e.stopPropagation();
          window.location.href = target;
        }
        return;
      }
      el = el.parentNode;
    }
  }, true);

  // Pass 3: keep rewriting as Squarespace renders and re-renders content.
  // Debounced so a busy page does not rescan on every mutation.
  function startObserver() {
    if (!window.MutationObserver || !document.body) return;
    var pending = false;
    var mo = new MutationObserver(function () {
      if (pending) return;
      pending = true;
      setTimeout(function () { pending = false; rewriteAll(); }, 50);
    });
    mo.observe(document.body, {
      childList: true, subtree: true,
      attributes: true, attributeFilter: ['href', 'action']
    });
  }

  function init() { rewriteAll(); startObserver(); }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
  // Catch anything that renders after DOMContentLoaded.
  window.addEventListener('load', rewriteAll);
})();
</script>

  var SVG_NS = 'http://www.w3.org/2000/svg';
  function el(tag, attrs, kids){
    var n = document.createElement(tag);
    if(attrs){ for(var k in attrs){ if(attrs.hasOwnProperty(k)){
      if(k === 'class') n.className = attrs[k];
      else if(k === 'html') n.innerHTML = attrs[k];
      else if(k === 'text') n.textContent = attrs[k];
      else n.setAttribute(k, attrs[k]);
    }}}
    if(kids){ (Array.isArray(kids)?kids:[kids]).forEach(function(c){
      if(c == null) return;
      n.appendChild(typeof c === 'string' ? document.createTextNode(c) : c);
    }); }
    return n;
  }
  function a(href, cls, kids, attrs){
    attrs = attrs || {}; attrs['class'] = cls; attrs.href = href;
    return el('a', attrs, kids);
  }

  function buildWordmark(){
    var L = CONFIG.brand.logo;
    if(L && L.src){
      var img = el('img', {
        'class':'sl-ap-logo-img' + (L.invertToBlack ? ' sl-ap-logo-invert' : ''),
        src:L.src, alt:CONFIG.brand.name, decoding:'async', loading:'eager'
      });

      if(L.heightMobile) img.style.setProperty('--sl-logo-h-m', L.heightMobile + 'px');
      var link = a(CONFIG.brand.home, 'sl-ap-wordmark sl-ap-wordmark-logo', [ img ], { 'aria-label':CONFIG.brand.name + ' home' });
      // graceful fallback to the text wordmark if the image ever fails to load
      img.addEventListener('error', function(){
        link.className = 'sl-ap-wordmark';
        link.innerHTML = '';
        link.appendChild(document.createTextNode(CONFIG.brand.wordmark[0]));
        link.appendChild(el('span',{ 'class':'sl-ap-dot','aria-hidden':'true' }));
        link.appendChild(document.createTextNode(CONFIG.brand.wordmark[1]));
      });
      return link;
    }
    return a(CONFIG.brand.home, 'sl-ap-wordmark', [
      CONFIG.brand.wordmark[0],
      el('span',{ 'class':'sl-ap-dot','aria-hidden':'true' }),
      CONFIG.brand.wordmark[1]
    ], { 'aria-label':CONFIG.brand.name + ' home' });
  }

  /* ════════════════════════════════════════════════════════════════

  function buildNav(){
    var nav = el('nav',{ 'class':'sl-ap-nav','aria-label':'Primary' });

    // Home — plain first link
    var homeWrap = el('span',{ 'class':'sl-ap-navwrap' });
    homeWrap.addEventListener('mouseenter', function(){
      closeDirectory();
      var ae = document.activeElement;
      if(ae && ae.classList && ae.classList.contains('sl-ap-navlink')) ae.blur();
    });
    homeWrap.appendChild(a(CONFIG.brand.home, 'sl-ap-navlink', ['Home']));
    nav.appendChild(homeWrap);

    // Our Apartments → opens the directory
    var resWrap = el('span',{ 'class':'sl-ap-navwrap' });
    var resLink = a('#', 'sl-ap-navlink sl-ap-has-panel', [CONFIG.brand.residencesLabel], {
      role:'button','aria-haspopup':'dialog','aria-expanded':'false','aria-controls':'sl-ap-dir'
    });
    resLink.addEventListener('click', function(e){
      e.preventDefault();
      if(body.classList.contains('sl-ap-dir-open')){ dirEl.focus(); }  // already hover-opened → move focus in
      else openDirectory();
    });
    resLink.addEventListener('keydown', function(e){
      if(e.key === 'Enter' || e.key === ' '){ e.preventDefault(); openDirectory(); }
    });
    /* hover opens the directory (desktop pointer devices only); blurring a
       click-focused sibling first so its dropdown can't hang open */
    resWrap.addEventListener('mouseenter', function(){
      var ae = document.activeElement;
      if(ae && ae.classList && ae.classList.contains('sl-ap-navlink') && ae !== resLink) ae.blur();
      if(hoverCapable && !isMobile()){ cancelDirClose(); openDirectory(true); }
    });
    resWrap.addEventListener('mouseleave', function(){
      if(hoverCapable && !isMobile()) armDirClose();
    });
    resWrap.appendChild(resLink);
    nav.appendChild(resWrap);
    residencesTrigger = resLink;

    // primary items — `links` → hover dropdown; `href` → plain top-level link
    CONFIG.primaryNav.forEach(function(item){
      var wrap = el('span',{ 'class':'sl-ap-navwrap' });

      /* one menu at a time: entering/focusing another nav item dismisses the
         Residences directory AND blurs a click-focused sibling nav link (the
         :focus-within reveal would otherwise pin its dropdown open — "hang"). */
      wrap.addEventListener('mouseenter', function(){
        closeDirectory();
        var ae = document.activeElement;
        if(ae && ae.classList && ae.classList.contains('sl-ap-navlink') && !wrap.contains(ae)) ae.blur();
      });
      wrap.addEventListener('focusin', function(){ closeDirectory(); });

      if(item.href){
        // plain link — no dropdown
        wrap.appendChild(a(item.href, 'sl-ap-navlink', [item.label]));
        nav.appendChild(wrap);
        return;
      }

      var link = a('#', 'sl-ap-navlink sl-ap-has-panel', [item.label], {
        'aria-haspopup':'true','aria-expanded':'false'
      });
      // a mouse click must not leave the link focused (keyboard Tab still reveals via :focus-within)
      link.addEventListener('click', function(e){ e.preventDefault(); link.blur(); });
      var drop = el('div',{ 'class':'sl-ap-drop', role:'menu','aria-label':item.label });
      drop.appendChild(el('p',{ 'class':'sl-ap-drop-label' }, item.label));
      item.links.forEach(function(l){
        drop.appendChild(a(l[1],'',[ l[0], el('span',{'aria-hidden':'true'},'→') ], { role:'menuitem' }));
      });
      wrap.appendChild(link); wrap.appendChild(drop);
      nav.appendChild(wrap);
    });

    return nav;
  }

  function openCityPanel(i){
    var p = cityPanelEls[i];
    if(!p) return;
    activeCityPanel = i;
    p.scrollTop = 0;
    p.classList.add('sl-ap-panel-open');
    if(cardEls[i]) cardEls[i].trigger.setAttribute('aria-expanded','true');
    var back = p.querySelector('.sl-ap-panel-back');
    if(back) requestAnimationFrame(function(){
      /* preventScroll: focusing inside the still-sliding panel must not
         side-scroll the curtain (it would stick shifted) */
      try{ back.focus({ preventScroll:true }); }catch(_){ back.focus(); }
      curtainEl.scrollLeft = 0;
      var sc = document.getElementById('sl-ap-curtain-scroll');
      if(sc) sc.scrollLeft = 0;
    });
  }
  function closeCityPanel(){
    if(activeCityPanel < 0) return;
    var i = activeCityPanel;
    var p = cityPanelEls[i];
    activeCityPanel = -1;
    if(p) p.classList.remove('sl-ap-panel-open');
    if(cardEls[i]){
      cardEls[i].trigger.setAttribute('aria-expanded','false');
      cardEls[i].trigger.focus();
    }
  }

  function focusables(c){
    return Array.prototype.slice.call(
      c.querySelectorAll('a[href], button:not([disabled]), summary, input, [tabindex]:not([tabindex="-1"])')
    ).filter(function(elm){
      if(elm.offsetParent === null) return false;
      // skip links inside a city panel that isn't slid in
      var panel = elm.closest ? elm.closest('.sl-ap-citypanel') : null;
      if(panel && !panel.classList.contains('sl-ap-panel-open')) return false;
      // skip links in the desktop facet sidebar while it's closed
      var right = elm.closest ? elm.closest('.sl-ap-col-right') : null;
      if(right){
        var d = document.getElementById('sl-ap-dir');
        if(d && !d.classList.contains('sl-ap-city-open')) return false;
      }
      return true;
    });
  }

    body.style.right = '';

<!--
  Student Luxe: site-wide image capper
  Paste into Squarespace: Settings > Advanced > Code Injection > HEADER.

  What it does, on every page, as the HTML is parsed:
    - finds hand-coded <img src="https://images.squarespace-cdn.com/..."> tags
      that carry no ?format= cap and no srcset (the gallery slides pasted
      into blog posts). Squarespace's own image blocks are left alone.
    - rewrites src to ?format=1000w and adds a 500 / 1000 / 1500 srcset with
      sizes, so phones get the small file and retina desktops the sharp one
    - lazy-loads every gallery slide except the first, async decode on all
    - PNGs (logos, icons) are skipped

  The data-sl-imgcap attribute is the marker the site-health crawler looks
  for: when it is present, uncapped tags are treated as handled.
-->
<script data-sl-imgcap="1">
(function () {
  var CDN = 'images.squarespace-cdn.com';
  var WIDTHS = [500, 1000, 1500];

  function fix(img) {
    if (img.__slCapped) return;
    var src = img.getAttribute('src') || '';
    if (src.indexOf(CDN) === -1) return;
    if (/[?&]format=/.test(src) || img.hasAttribute('srcset')) return;
    if (/\.png(\?|$)/i.test(src)) return;
    var base = src.split('?')[0];
    img.setAttribute('src', base + '?format=1000w');
    img.setAttribute('srcset', WIDTHS.map(function (w) { return base + '?format=' + w + 'w ' + w + 'w'; }).join(', '));
    img.setAttribute('sizes', '(max-width: 640px) 100vw, 1000px');
    if (!img.hasAttribute('loading')) {
      var slide = img.closest ? img.closest('.slide') : null;
      var first = !slide || !slide.previousElementSibling;
      img.setAttribute('loading', first ? 'eager' : 'lazy');
    }
    if (!img.hasAttribute('decoding')) img.setAttribute('decoding', 'async');
    img.__slCapped = true;
  }

  function sweep(node) {
    if (node.tagName === 'IMG') { fix(node); return; }
    if (!node.querySelectorAll) return;
    var imgs = node.querySelectorAll('img');
    for (var i = 0; i < imgs.length; i++) fix(imgs[i]);
  }

  try {
    new MutationObserver(function (muts) {
      for (var i = 0; i < muts.length; i++) {
        var added = muts[i].addedNodes;
        for (var j = 0; j < added.length; j++) if (added[j].nodeType === 1) sweep(added[j]);
      }
    }).observe(document.documentElement, { childList: true, subtree: true });
  } catch (e) { /* very old browser: the DOMContentLoaded sweep still runs */ }

  document.addEventListener('DOMContentLoaded', function () { sweep(document); });
})();
</script>

    '<path fill="#4285F4" d="M45.12 24.5c0-1.56-.14-3.06-.4-4.5H24v8.51h11.84c-.51 2.75-2.06 5.08-4.39 6.64v5.52h7.11c4.16-3.83 6.56-9.47 6.56-16.17z"/>' +
    '<path fill="#34A853" d="M24 46c5.94 0 10.92-1.97 14.56-5.33l-7.11-5.52c-1.97 1.32-4.49 2.1-7.45 2.1-5.73 0-10.58-3.87-12.31-9.07H4.34v5.7C7.96 41.07 15.4 46 24 46z"/>' +
    '<path fill="#FBBC05" d="M11.69 28.18C11.25 26.86 11 25.45 11 24s.25-2.86.69-4.18v-5.7H4.34C2.85 17.09 2 20.45 2 24c0 3.55.85 6.91 2.34 9.88l7.35-5.7z"/>' +
    '<path fill="#EA4335" d="M24 10.75c3.23 0 6.13 1.11 8.41 3.29l6.31-6.31C34.91 4.18 29.93 2 24 2 15.4 2 7.96 6.93 4.34 14.12l7.35 5.7C13.42 14.62 18.27 10.75 24 10.75z"/></svg>';
  var CHEVRON = '<svg viewBox="0 0 10 10"><polyline points="2,3.5 5,6.5 8,3.5"/></svg>';

  function esc(s) { return String(s).replace(/[&<>"']/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]; }); }

  function slideHtml(r, i, total) {
    var name = esc(r.name);
    return '<div class="slx-rev-slide" role="group" aria-roledescription="slide" aria-label="' + (i + 1) + ' of ' + total + '">' +
      '<div class="slx-rev-stars" aria-label="5 out of 5 stars">' + STAR + STAR + STAR + STAR + STAR + '</div>' +
      '<div class="slx-rev-quote-wrap"><p class="slx-rev-quote">' + esc(r.quote) + '</p></div>' +
      '<button class="slx-rev-toggle" type="button" aria-expanded="false"><span class="slx-rev-toggle-label">Read more</span>' + CHEVRON + '</button>' +
      '<div class="slx-rev-attr"><div class="slx-rev-avatar">' + esc(r.initials) + '</div>' +
      '<div class="slx-rev-attr-text"><span class="slx-rev-name">' + name + '</span>' +
      '<a class="slx-rev-source" href="' + esc(r.url) + '" target="_blank" rel="noopener" aria-label="Read ' + name + '’s verified review on Google">' + GOOGLE_G + ' Verified review on Google &rarr;</a>' +
      '</div></div></div>';
  }

  function build(root) {
    var total = REVIEWS.length;
    root.innerHTML =
      '<button class="slx-rev-arrow prev" type="button" aria-label="Previous review"><svg viewBox="0 0 14 14"><polyline points="9,2 4,7 9,12"/></svg></button>' +
      '<div class="slx-rev-viewport"><div class="slx-rev-track">' + REVIEWS.map(function (r, i) { return slideHtml(r, i, total); }).join('') + '</div></div>' +
      '<button class="slx-rev-arrow next" type="button" aria-label="Next review"><svg viewBox="0 0 14 14"><polyline points="5,2 10,7 5,12"/></svg></button>' +
      '<div class="slx-rev-dots" role="tablist" aria-label="Select review">' + REVIEWS.map(function (_, i) { return '<button class="slx-rev-dot' + (i ? '' : ' active') + '" data-idx="' + i + '" aria-label="Review ' + (i + 1) + '" type="button"></button>'; }).join('') + '</div>' +
      '<a href="/our-reviews" class="slx-rev-all">Read all reviews &rarr;</a>';

    var viewport = root.querySelector('.slx-rev-viewport');
    var track = root.querySelector('.slx-rev-track');
    var dots = root.querySelectorAll('.slx-rev-dot');
    var slides = root.querySelectorAll('.slx-rev-slide');
    var current = 0, timer = null, AUTOPLAY_MS = 6500;
    var reduced = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    function collapseAll() {
      root.querySelectorAll('.slx-rev-quote-wrap.expanded').forEach(function (w) {
        w.classList.remove('expanded');
        var btn = w.parentNode.querySelector('.slx-rev-toggle');
        if (btn) { btn.setAttribute('aria-expanded', 'false'); var l = btn.querySelector('.slx-rev-toggle-label'); if (l) l.textContent = 'Read more'; }
      });
    }
    function setIndex(i) {
      current = (i + total) % total;
      track.style.transform = 'translateX(' + (-current * 100) + '%)';
      dots.forEach(function (d, k) { d.classList.toggle('active', k === current); });
      collapseAll();
    }
    function next() { setIndex(current + 1); }
    function prev() { setIndex(current - 1); }
    function stop() { if (timer) { clearInterval(timer); timer = null; } }
    function start() { if (reduced) return; stop(); timer = setInterval(next, AUTOPLAY_MS); }
    function restart() { stop(); start(); }

    function fitToggles() {
      slides.forEach(function (s) {
        var wrap = s.querySelector('.slx-rev-quote-wrap'), q = s.querySelector('.slx-rev-quote');
        if (!wrap || !q || wrap.classList.contains('expanded')) return;
        wrap.classList.toggle('no-overflow', q.scrollHeight <= q.clientHeight + 2);
      });
    }
    slides.forEach(function (s) {
      var wrap = s.querySelector('.slx-rev-quote-wrap'), btn = s.querySelector('.slx-rev-toggle');
      btn.addEventListener('click', function (e) {
        e.preventDefault(); e.stopPropagation();
        var open = wrap.classList.toggle('expanded');
        btn.setAttribute('aria-expanded', open ? 'true' : 'false');
        var l = btn.querySelector('.slx-rev-toggle-label'); if (l) l.textContent = open ? 'Read less' : 'Read more';
        if (open) stop(); else restart();
      });
    });
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(fitToggles); else setTimeout(fitToggles, 100);
    var rt; window.addEventListener('resize', function () { clearTimeout(rt); rt = setTimeout(function () { requestAnimationFrame(fitToggles); }, 200); });

    dots.forEach(function (d) { d.addEventListener('click', function () { setIndex(parseInt(d.getAttribute('data-idx'), 10)); restart(); }); });
    root.querySelector('.slx-rev-arrow.prev').addEventListener('click', function () { prev(); restart(); });
    root.querySelector('.slx-rev-arrow.next').addEventListener('click', function () { next(); restart(); });
    root.addEventListener('mouseenter', stop);
    root.addEventListener('mouseleave', start);
    if ('IntersectionObserver' in window) {
      new IntersectionObserver(function (entries) { entries.forEach(function (e) { if (e.isIntersecting) start(); else stop(); }); }, { threshold: 0.2 }).observe(root);
    } else { start(); }

    var startX = 0, deltaX = 0, dragging = false;
    viewport.addEventListener('pointerdown', function (e) {
      if (e.target && e.target.closest && e.target.closest('a, button')) return;
      if (root.querySelector('.slx-rev-quote-wrap.expanded')) return;
      dragging = true; startX = e.clientX; deltaX = 0; track.classList.add('no-anim');
      try { viewport.setPointerCapture(e.pointerId); } catch (_) {}
      stop();
    });
    viewport.addEventListener('pointermove', function (e) {
      if (!dragging) return;
      deltaX = e.clientX - startX;
      track.style.transform = 'translateX(' + (-current * 100 + (deltaX / viewport.offsetWidth) * 100) + '%)';
    });
    function up() {
      if (!dragging) return;
      dragging = false; track.classList.remove('no-anim');
      if (Math.abs(deltaX) > 50) setIndex(current + (deltaX < 0 ? 1 : -1)); else setIndex(current);
      deltaX = 0; start();
    }
    viewport.addEventListener('pointerup', up);
    viewport.addEventListener('pointercancel', up);
    viewport.addEventListener('pointerleave', function () { if (dragging) up(); });
    viewport.addEventListener('dragstart', function (e) { e.preventDefault(); });
    root.setAttribute('tabindex', '0');
    root.addEventListener('keydown', function (e) {
      if (e.key === 'ArrowLeft') { prev(); restart(); } else if (e.key === 'ArrowRight') { next(); restart(); }
    });
  }

  function swap(el) {
    if (el.__slSwapped) return;
    el.__slSwapped = true;
    var prevEl = el.previousElementSibling;
    if (prevEl && prevEl.tagName === 'SCRIPT' && /static\.elfsight\.com\/platform/.test(prevEl.src || '')) prevEl.parentNode.removeChild(prevEl);
    var root = document.createElement('div');
    root.className = 'slx-rev';
    root.setAttribute('aria-roledescription', 'carousel');
    root.setAttribute('aria-label', 'Customer reviews');
    el.parentNode.replaceChild(root, el);
    build(root);
  }

  function sweep(node) {
    if (node.nodeType !== 1) return;
    if (node.classList && node.classList.contains(WIDGET)) { swap(node); return; }
    if (!node.querySelectorAll) return;
    var found = node.querySelectorAll('.' + WIDGET);
    for (var i = 0; i < found.length; i++) swap(found[i]);
  }

  try {
    new MutationObserver(function (muts) {
      for (var i = 0; i < muts.length; i++) {
        var added = muts[i].addedNodes;
        for (var j = 0; j < added.length; j++) sweep(added[j]);
      }
    }).observe(document.documentElement, { childList: true, subtree: true });
  } catch (e) { /* the DOMContentLoaded sweep still runs */ }
  document.addEventListener('DOMContentLoaded', function () { sweep(document.documentElement); });
  window.slRevSwap = { sweep: function () { sweep(document.documentElement); } };
})();
</script>
