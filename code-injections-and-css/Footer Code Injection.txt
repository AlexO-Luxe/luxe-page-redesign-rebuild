<script>
(function() {
  try {
    var KEY      = 'sl_visited_paths';
    var SRC_KEY  = 'sl_traffic_source';
    var TS_KEY   = 'sl_page_arrived';
    var MAX_PAGES = 30;
    var path     = window.location.pathname.replace(/\/$/, '') || '/';

    // Skip asset paths
    if (/\.(js|css|png|jpg|jpeg|gif|svg|ico|woff|woff2|ttf|pdf)$/i.test(path)) return;

    var now      = Date.now();
    var params   = new URLSearchParams(window.location.search);
    var referrer = document.referrer || '';

    // Detect traffic source (only on first page of session)
    function detectSource() {
      var gclid   = params.get('gclid');
      var fbclid  = params.get('fbclid');
      var source  = (params.get('utm_source') || '').toLowerCase();
      var medium  = (params.get('utm_medium') || '').toLowerCase();

      if (gclid || (source.indexOf('google') > -1 && (medium === 'cpc' || medium === 'ppc'))) return 'Google Ads';
      if (fbclid || source.indexOf('facebook') > -1 || source.indexOf('meta') > -1) return 'Meta Ads';
      if (source.indexOf('instagram') > -1) return 'Instagram';
      if (source.indexOf('bing') > -1 && (medium === 'cpc' || medium === 'ppc')) return 'Bing Ads';
      if (source.indexOf('tiktok') > -1) return 'TikTok';
      if (source && medium === 'email') return 'Email';
      if (source) return source.charAt(0).toUpperCase() + source.slice(1);

      if (referrer) {
        try {
          var ref = new URL(referrer).hostname.replace('www.', '');
          if (ref.indexOf('google.') > -1)    return 'Google Organic';
          if (ref.indexOf('bing.com') > -1)   return 'Bing Organic';
          if (ref.indexOf('yahoo.com') > -1)  return 'Yahoo Organic';
          if (ref.indexOf('facebook.com') > -1) return 'Facebook';
          if (ref.indexOf('instagram.com') > -1) return 'Instagram';
          if (ref.indexOf('linkedin.com') > -1) return 'LinkedIn';
          if (ref.indexOf('twitter.com') > -1 || ref.indexOf('x.com') > -1) return 'X / Twitter';
          if (ref.indexOf('tiktok.com') > -1) return 'TikTok';
          if (ref.indexOf('studentluxe.co.uk') > -1) return null;
          return 'Referral: ' + ref;
        } catch(e) {}
      }

      return 'Direct';
    }

    var existing    = localStorage.getItem(KEY) || '';
    var pages       = existing ? existing.split(' 👉 ') : [];
    var lastArrived = parseInt(localStorage.getItem(TS_KEY) || '0', 10);
    var isFirst     = pages.length === 0;

    if (isFirst) {
      var source = detectSource();
      if (source) {
        localStorage.setItem(SRC_KEY, source);
        pages.push(source);
      }
    }

    var hasContent  = pages.filter(function(p) { return p.indexOf('/') > -1; }).length === 0;
    var entry       = hasContent
      ? (window.location.hostname + window.location.pathname + window.location.search)
      : path;

    if (pages.length > 0 && lastArrived > 0) {
      var last = pages[pages.length - 1];
      if (last.indexOf('/') > -1 && last.indexOf('(') === -1) {
        var elapsed = Math.round((now - lastArrived) / 1000);
        var mins    = Math.floor(elapsed / 60);
        var secs    = elapsed % 60;
        var timeStr = mins > 0
          ? mins + ':' + (secs < 10 ? '0' : '') + secs + 'm'
          : secs + 's';
        pages[pages.length - 1] = last + ' (' + timeStr + ')';
      }
    }

    var lastEntry = pages.length > 0 ? pages[pages.length - 1] : '';
    var lastPath  = lastEntry.replace(/ \(.*\)$/, '').split('?')[0];
    var thisClean = window.location.hostname + path;

    if (lastPath !== path && lastPath !== thisClean) {
      pages.push(entry);
      if (pages.length > MAX_PAGES) pages = pages.slice(-MAX_PAGES);
      localStorage.setItem(KEY, pages.join(' 👉 '));
    }

    localStorage.setItem(TS_KEY, now.toString());

  } catch(e) {}
})();
</script>

<script>
(function () {
  'use strict';

  var COOKIE_DAYS = 90;
  var TRACK_URL   = 'https://luxe-lead-capture.vercel.app/api/track';

  function setCookie(name, value, days) {
    if (value == null || value === '') return;
    var expires = new Date(Date.now() + days * 864e5).toUTCString();
    document.cookie = name + '=' + encodeURIComponent(value) +
      '; expires=' + expires + '; path=/; SameSite=Lax; Secure';
  }
  function getCookie(name) {
    var match = document.cookie.match(new RegExp('(^|; )' + name + '=([^;]*)'));
    return match ? decodeURIComponent(match[2]) : '';
  }
  function setFirstTouch(name, value) {
    if (!getCookie(name) && value) setCookie(name, value, COOKIE_DAYS);
  }
  function uuid() {
    if (window.crypto && crypto.randomUUID) return crypto.randomUUID();
    return 'sl-' + Date.now().toString(36) + '-' +
      Math.random().toString(36).slice(2, 10);
  }

  // Session ID
  if (!getCookie('sl_session_id')) setCookie('sl_session_id', uuid(), COOKIE_DAYS);
  if (!getCookie('sl_first_seen')) setCookie('sl_first_seen', new Date().toISOString(), COOKIE_DAYS);

  // URL params to cookies
  var params = new URLSearchParams(window.location.search);
  var fields = ['gclid', 'gbraid', 'wbraid', 'fbclid',
                'utm_campaign', 'utm_adgroup', 'utm_term',
                'utm_matchtype', 'utm_source', 'utm_medium'];

  fields.forEach(function (f) {
    var v = params.get(f);
    if (!v) return;
    var key;
    if (f.indexOf('utm_') === 0) {
      key = (f === 'utm_source' || f === 'utm_medium') ? 'sl_' + f : 'sl_' + f.slice(4);
    } else {
      key = 'sl_' + f;
    }
    setCookie(key, v, COOKIE_DAYS);
    setFirstTouch('sl_first_' + key.slice(3), v);
  });

  // First-touch referrer + landing (skip same-host referrers)
  if (document.referrer && document.referrer.indexOf(location.host) === -1) {
    setFirstTouch('sl_first_referrer', document.referrer);
    setCookie('sl_referrer', document.referrer, COOKIE_DAYS);
  }
  setFirstTouch('sl_first_landing_page', location.pathname);
  setCookie('sl_landing_page', location.pathname, COOKIE_DAYS);

  // Page-view ping (fire-and-forget). keepalive survives unload.
  try {
    fetch(TRACK_URL, {
      method:      'POST',
      credentials: 'omit',
      keepalive:   true,
      headers:     { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        session_id:    getCookie('sl_session_id'),
        gclid:         getCookie('sl_gclid'),
        gbraid:        getCookie('sl_gbraid'),
        wbraid:        getCookie('sl_wbraid'),
        fbclid:        getCookie('sl_fbclid'),
        utm_campaign:  getCookie('sl_campaign'),
        utm_adgroup:   getCookie('sl_adgroup'),
        utm_term:      getCookie('sl_term'),
        utm_matchtype: getCookie('sl_matchtype'),
        utm_source:    getCookie('sl_utm_source'),
        utm_medium:    getCookie('sl_utm_medium'),
        referrer:      document.referrer || '',
        landing_page:  getCookie('sl_first_landing_page'),
        path:          location.pathname
      })
    });
  } catch (e) {}

  // Mirror cookies into hidden form inputs on every form submit.
  // visited_paths intentionally omitted: the path-tracker above owns that key.
  function mirrorIntoForms() {
    var forms = document.querySelectorAll('form');
    forms.forEach(function (form) {
      if (form.dataset.slMirrored === '1') return;
      form.dataset.slMirrored = '1';
      var cookieMap = {
        session_id:         'sl_session_id',
        gclid:              'sl_gclid',
        gbraid:             'sl_gbraid',
        wbraid:             'sl_wbraid',
        fbclid:             'sl_fbclid',
        utm_campaign:       'sl_campaign',
        utm_adgroup:        'sl_adgroup',
        utm_term:           'sl_term',
        utm_matchtype:      'sl_matchtype',
        utm_source:         'sl_utm_source',
        utm_medium:         'sl_utm_medium',
        referrer:           'sl_referrer',
        landing_page:       'sl_landing_page',
        first_gclid:        'sl_first_gclid',
        first_gbraid:       'sl_first_gbraid',
        first_wbraid:       'sl_first_wbraid',
        first_campaign:     'sl_first_campaign',
        first_adgroup:      'sl_first_adgroup',
        first_term:         'sl_first_term',
        first_referrer:     'sl_first_referrer',
        first_landing_page: 'sl_first_landing_page',
        first_seen:         'sl_first_seen'
      };
      Object.keys(cookieMap).forEach(function (name) {
        if (form.querySelector('input[name="' + name + '"]')) return;
        var input = document.createElement('input');
        input.type  = 'hidden';
        input.name  = name;
        input.value = getCookie(cookieMap[name]) || '';
        form.appendChild(input);
      });
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', mirrorIntoForms);
  } else {
    mirrorIntoForms();
  }
  var mo = new MutationObserver(mirrorIntoForms);
  mo.observe(document.body, { childList: true, subtree: true });
})();
</script>

// Global Pricing Dropdown Initializer
// Add this to Settings → Advanced → Code Injection → Footer

(function() {
  function formatPrice(price, currency, multiplier = 1, pricingMode = 'weekly') {
    if (!price && price !== 0) return 'TBA';

    const priceStr = String(price);

    if (priceStr.includes(currency) || priceStr.includes('per')) {
      return priceStr;
    }

    let num = parseFloat(price);
    if (!isNaN(num)) {
      num = num * multiplier;
      num = Math.ceil(num / 5) * 5;
      const formatted = num.toLocaleString('en-GB');
      const period = pricingMode === 'monthly' ? 'monthly' : 'per week';
      return `${currency}${formatted} ${period}`;
    }

    return priceStr;
  }

  function initializePricingDropdowns() {
    const dropdownContainers = document.querySelectorAll('.pricing-dropdown');

    dropdownContainers.forEach(container => {
      const dropdown = container.querySelector('.stay-duration-select');
      const priceDisplay = container.querySelector('.stay-price');

      if (!dropdown || !priceDisplay) {
        console.warn('Pricing dropdown structure not found in container:', container);
        return;
      }

      const currency = container.getAttribute('data-currency') || '£';

      const pricingMode = container.getAttribute('data-pricing') || 'weekly';
      const pricingMultiplier = pricingMode === 'monthly' ? 4.3 : 1;

      const defaultSelection = container.getAttribute('data-default') || null;

      const prices = {
        long: '',
        medium: '',
        short: '',
        summer: ''
      };

      function getPropertyRef() {
        return container.getAttribute('data-property-ref');
      }

      async function fetchPricing() {
        try {
          const propertyRef = getPropertyRef();

          if (!propertyRef) {
            priceDisplay.innerHTML = "<i>Unable to load pricing</i>";
            dropdown.disabled = true;
            return;
          }

          const apiUrl = `https://apartment-pricing-api.vercel.app/api/pricing?property_ref=${encodeURIComponent(propertyRef)}`;

          const response = await fetch(apiUrl);

          if (!response.ok) {
            throw new Error(`API error: ${response.status}`);
          }

          const data = await response.json();

          const longRate = parseFloat(data.long);

          let mediumRate = data.medium ? parseFloat(data.medium) : (longRate * 1.20);
          let shortRate = data.short ? parseFloat(data.short) : (longRate * 1.40);
          let summerRate = data.summer ? parseFloat(data.summer) : (longRate * 1.50);

          prices.long = formatPrice(longRate, currency, pricingMultiplier, pricingMode);
          prices.medium = formatPrice(mediumRate, currency, pricingMultiplier, pricingMode);
          prices.short = formatPrice(shortRate, currency, pricingMultiplier, pricingMode);
          prices.summer = formatPrice(summerRate, currency, pricingMultiplier, pricingMode);

          dropdown.dataset.long = prices.long;
          dropdown.dataset.medium = prices.medium;
          dropdown.dataset.short = prices.short;
          dropdown.dataset.summer = prices.summer;

          dropdown.disabled = false;
          priceDisplay.innerHTML = "<i>Prices based on length of stay</i>";

          if (defaultSelection) {
            dropdown.value = defaultSelection;
            updatePrice();
          }

          if (document.referrer.includes("/lse-summer-accommodation")) {
            dropdown.value = "summer";
            updatePrice();
          }

        } catch (error) {
          console.error('Error fetching pricing:', error);
          priceDisplay.innerHTML = "<i>Unable to load prices</i>";
          dropdown.disabled = true;
        }
      }

      function updatePrice() {
        if (dropdown.value && prices[dropdown.value]) {
          priceDisplay.innerHTML = `from &nbsp;<span class="stay-price-amount">${prices[dropdown.value]}</span>`;
        } else {
          priceDisplay.innerHTML = "<i>Please select your stay length</i>";
        }
      }

      dropdown.addEventListener("change", () => {
        priceDisplay.style.opacity = 0;
        setTimeout(() => {
          updatePrice();
          priceDisplay.style.opacity = 1;
        }, 300);
      });

      fetchPricing();
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initializePricingDropdowns);
  } else {
    initializePricingDropdowns();
  }
})();
</script>

    if (e.target && e.target.id === 'sl-cb-close') closeSheet();
  });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && sheet.classList.contains('sl-open')) closeSheet();
  });
  document.getElementById('sl-cb-x').addEventListener('click', function (e) {
    e.stopPropagation();
    closeSheet();
    wrap.style.display = 'none';
  });

  /* ── context wiring ── */
  function escHtml(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  function prettyCity(c) {
    return String(c || '').replace(/-/g, ' ').replace(/\b\w/g, function (ch) { return ch.toUpperCase(); });
  }

  function detectCityFromPath() {
    var path = window.location.pathname.toLowerCase();
    var cities = ['london', 'paris', 'new-york', 'cambridge', 'madrid', 'edinburgh', 'milan', 'boston', 'chicago', 'washington', 'los-angeles'];
    for (var i = 0; i < cities.length; i++) {
      if (path.indexOf(cities[i]) !== -1) return cities[i];
    }
    return null;
  }

  function cleanName(name) {
    return name
      .replace(/\s+(studio|penthouse|\d[\s-]*bed(room)?s?|one[\s-]*bed(room)?|deluxe[\s-]*(studio|\d[\s-]*bed)|executive[\s-]*studio|accessible[\s-]*studio|double[\s-]*studio|twin[\s-]*bed|small[\s-]*double.*|w\/\s*patio.*)$/i, '')
      .trim();
  }

  function updateContext() {
    var aRow    = document.getElementById('sl-cb-a');
    var aDesc   = document.getElementById('sl-cb-a-d');
    var bRow    = document.getElementById('sl-cb-b');
    var bTitle  = document.getElementById('sl-cb-b-t');

    var pathCity = detectCityFromPath();
    var city = currentApt ? currentApt.city : pathCity;
    var cityLabel = city ? prettyCity(city) : '';

    if (currentApt) {
      var displayName = cleanName(currentApt.name);
      aDesc.innerHTML = 'Submit your desired dates for <b>' + escHtml(displayName) + '</b>';
      aRow.href = FORM_URL + '?type=A&ref=' + encodeURIComponent(displayName) + '&city=' + encodeURIComponent(currentApt.city);
    } else {
      aDesc.textContent = 'Enquire about a specific apartment';
      aRow.href = FORM_URL + '?type=A';
    }

    bTitle.innerHTML = cityLabel ? 'Send me <b>' + escHtml(cityLabel) + '</b> options' : 'Send me options';
    bRow.href = FORM_URL + '?type=B' + (city ? '&city=' + encodeURIComponent(city) : '');

    /* desktop-only pill chip: apartment name (first two words) beats city */
    var chip = document.getElementById('sl-cb-chip');
    if (chip) {
      var chipText = '';
      if (currentApt) chipText = cleanName(currentApt.name).split(/\s+/).slice(0, 2).join(' ');
      else if (cityLabel) chipText = cityLabel;
      chip.textContent = chipText;
      chip.classList.toggle('sl-on', !!chipText);
    }
  }

  function swapJotformButtons(apt) {
    document.querySelectorAll('a[href*="jotform"]').forEach(function (el) {
      if (apt) {
        var displayName = cleanName(apt.name);
        el.href = FORM_URL + '?type=A&ref=' + encodeURIComponent(displayName) + '&city=' + encodeURIComponent(apt.city);
      } else {
        var city = detectCityFromPath();
        el.href = FORM_URL + '?type=B' + (city ? '&city=' + encodeURIComponent(city) : '');
      }
      el.removeAttribute('target');
      el.removeAttribute('rel');
    });
  }

  function matchCurrentPage() {
    var path = window.location.pathname.replace(/\/$/, '').toLowerCase();
    if (aptMap[path]) {
      currentApt = aptMap[path];
      updateContext();
      swapJotformButtons(currentApt);
    } else {
      updateContext();
      swapJotformButtons(null);
    }
  }

  updateContext();

  fetch(API_URL)
    .then(function (r) { return r.json(); })
    .then(function (data) {
      if (!data.apartments) return;
      data.apartments.forEach(function (apt) {
        aptMap[apt.slug] = { name: apt.name, city: apt.city };
      });
      matchCurrentPage();
    })
    .catch(function (err) {
      console.warn('[SL Widget] Could not load apartment map:', err);
      swapJotformButtons(null);
    });

})();
</script>

(function () {
  var API_URL = 'https://luxe-lead-capture.vercel.app/api/submit-whatsapp';
  var DEFAULT_NUMBER = '442030070017';
  var API_TIMEOUT_MS = 2500;

  var overlay = document.getElementById('slwaOverlay');
  var nameInput = document.getElementById('slwaName');
  var needInput = document.getElementById('slwaNeed');
  var goBtn = document.getElementById('slwaGo');
  var pendingNumber = DEFAULT_NUMBER;

  // Localised opening hours: 10:00 to 18:00 London time, shown in the
  // visitor's own timezone (falls back to the plain GMT line).
  try {
    var visitorZone = Intl.DateTimeFormat().resolvedOptions().timeZone || '';
    if (visitorZone && visitorZone !== 'Europe/London') {
      var fmtLondon = new Intl.DateTimeFormat('en-GB', { timeZone: 'Europe/London', hour: 'numeric', minute: 'numeric', hour12: false });
      var fmtLocal = new Intl.DateTimeFormat('en-GB', { hour: 'numeric', minute: 'numeric', hour12: true, timeZoneName: 'short' });
      var toLocal = function (h) {
        var d = new Date();
        var parts = fmtLondon.format(d).split(':');
        d.setMinutes(d.getMinutes() + ((h * 60) - (parseInt(parts[0], 10) * 60 + parseInt(parts[1], 10))));
        return fmtLocal.format(d).replace(/\s?([ap]m)/, '$1');
      };
      document.getElementById('slwaHours').textContent =
        "We're live from " + toLocal(10) + ' to ' + toLocal(18) + ' your time.';
    }
  } catch (e) { /* keep the GMT default */ }

  function getCookie(name) {
    var m = document.cookie.match('(?:^|; )' + name + '=([^;]*)');
    if (m) return decodeURIComponent(m[1]);
    try { return localStorage.getItem('ct_' + name.replace(/^sl_/, '')) || ''; } catch (e) { return ''; }
  }

  function trackingData() {
    return {
      timestamp: new Date().toISOString(),
      page_path: location.pathname + location.search,
      last_page: getCookie('sl_last_page'),
      landing_page: getCookie('sl_landing_page'),
      utm_source: getCookie('sl_utm_source'),
      utm_medium: getCookie('sl_utm_medium'),
      utm_campaign: getCookie('sl_campaign') || getCookie('sl_utm_campaign'),
      utm_adgroup: getCookie('sl_adgroup'),
      utm_term: getCookie('sl_term'),
      utm_matchtype: getCookie('sl_matchtype'),
      gclid: getCookie('sl_gclid'),
      gbraid: getCookie('sl_gbraid'),
      wbraid: getCookie('sl_wbraid'),
      fbclid: getCookie('sl_fbclid'),
      session_id: getCookie('sl_session_id'),
      // The page tracker keeps the journey in localStorage under its own key,
      // not in a cookie, so read it from there first.
      visited_paths: (function () { try { return localStorage.getItem('sl_visited_paths') || getCookie('sl_visited_paths'); } catch (e) { return getCookie('sl_visited_paths'); } })(),
      first_gclid: getCookie('sl_first_gclid'),
      first_campaign: getCookie('sl_first_campaign'),
      referrer: getCookie('sl_referrer'),
      first_referrer: getCookie('sl_first_referrer'),
      first_seen: getCookie('sl_first_seen'),
      traffic_source: (function () { try { return localStorage.getItem('sl_traffic_source') || ''; } catch (e) { return ''; } })(),
      page_arrived: (function () { try { return localStorage.getItem('sl_page_arrived') || ''; } catch (e) { return ''; } })()
    };
  }

  function extractNumber(href) {
    var m = (href || '').match(/(?:wa\.me\/|phone=)\+?(\d{6,15})/);
    return m ? m[1] : DEFAULT_NUMBER;
  }

  function openModal(number) {
    pendingNumber = number || DEFAULT_NUMBER;
    overlay.classList.add('slwa-open');
    try { nameInput.value = nameInput.value || localStorage.getItem('slwa_name') || ''; } catch (e) {}
    (nameInput.value ? needInput : nameInput).focus();
  }

  function closeModal() {
    overlay.classList.remove('slwa-open');
    goBtn.disabled = false;
    goBtn.lastChild.textContent = ' Open WhatsApp';
  }

  document.getElementById('slwaClose').addEventListener('click', closeModal);
  overlay.addEventListener('click', function (e) { if (e.target === overlay) closeModal(); });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && overlay.classList.contains('slwa-open')) closeModal();
  });

  document.addEventListener('click', function (e) {
    var a = e.target && e.target.closest ? e.target.closest('a[href]') : null;
    if (!a) return;
    var href = a.getAttribute('href') || '';
    if (a.hasAttribute('data-slwa-skip')) return;
    if (!/wa\.me\/|api\.whatsapp\.com|whatsapp\.com\/send|^whatsapp:/i.test(href)) return;
    e.preventDefault();
    e.stopPropagation();
    openModal(extractNumber(href));
  }, true);

  function withTimeout(promise, ms) {
    return Promise.race([promise, new Promise(function (resolve) { setTimeout(function () { resolve(null); }, ms); })]);
  }

  goBtn.addEventListener('click', function () {
    var name = nameInput.value.trim();
    var need = needInput.value.trim();
    nameInput.classList.toggle('slwa-error', !name);

      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    }).then(function (r) { return r.json(); }).catch(function () { return null; });

    withTimeout(post, API_TIMEOUT_MS).then(function (data) {
